import {readFileSync,writeFileSync,mkdirSync} from 'node:fs';
import {rm} from 'node:fs/promises';
import {join} from 'node:path';
import Database from '/workspace/code/node_modules/better-sqlite3/lib/index.js';
import {createEncryptedBackup,restoreEncryptedBackup} from '/workspace/code/modules/learning-state/backup';
import {createDiscoveryBackup,restoreDiscoveryBackup} from '/workspace/code/modules/discovery/backup';
const plan=JSON.parse(readFileSync('/work/backup-plan.json','utf8')),passphrase=process.env.BACKUP_PASSPHRASE;const result={pairs:[],singles:[],totalDatabases:plan.totalDatabases,allVerified:false};
function counts(filename){const db=new Database(filename,{readonly:true,fileMustExist:true});try{if(db.pragma('quick_check',{simple:true})!=='ok')throw new Error('integrity failed');return Object.fromEntries(db.prepare("SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' ORDER BY name").all().map(({name})=>[name,db.prepare('SELECT COUNT(*) AS n FROM "'+name.replaceAll('"','""')+'"').get().n]));}finally{db.close();}}
try{for(const item of plan.pairs){const b=await createDiscoveryBackup({stateFilename:item.state,discoveryFilename:item.discovery,outputDirectory:'/secure/paired',passphrase,appVersion:'pre-course-navigation-20261115'});const target='/secure/restore-checks/'+b.batch.batchId;await restoreDiscoveryBackup({batchFilename:join(b.directory,'batch.json'),targetDirectory:target,passphrase});await rm(target,{recursive:true});result.pairs.push({label:item.label,batchId:b.batch.batchId,verified:true});console.log('Verified paired backup: '+item.label);}
for(const item of plan.singles){const outputDirectory='/secure/single/'+item.label;const before=counts(item.source);const b=await createEncryptedBackup({sourceFilename:item.source,outputDirectory,passphrase});const target='/secure/restore-checks/'+item.label+'.sqlite';await restoreEncryptedBackup({backupFilename:b.backupPath,targetFilename:target,passphrase});if(JSON.stringify(before)!==JSON.stringify(counts(target)))throw new Error('restored count mismatch');await rm(target);result.singles.push({label:item.label,backup:b.manifest.filename,verified:true});console.log('Verified single backup: '+item.label);}
result.allVerified=true;}finally{writeFileSync('/secure/backup-verification.json',JSON.stringify(result,null,2)+'\n',{mode:0o600});writeFileSync('/reports/backup-verification.json',JSON.stringify(result,null,2)+'\n');}
