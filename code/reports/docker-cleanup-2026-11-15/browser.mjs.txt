import {chromium} from '/Users/vsh9p8q/AI/Agent-Learning-Hub/code/node_modules/playwright/index.mjs';
import {mkdir,writeFile} from 'node:fs/promises';
const root='/Users/vsh9p8q/AI/Agent-Learning-Hub/code/reports/docker-cleanup-2026-11-15';await mkdir(root+'/screenshots',{recursive:true});
const browser=await chromium.launch({channel:'chrome'});const results=[];
try{for(const mode of ['local','cloud']){const base=mode==='local'?'http://127.0.0.1:3003':'http://127.0.0.1:3002';const page=await browser.newPage({viewport:{width:1440,height:1000},locale:'zh-CN'});const errors=[];page.on('pageerror',e=>errors.push(e.message));const response=await page.goto(base,{waitUntil:'domcontentloaded'});if(!response.ok())throw new Error(mode+' home unavailable');const badge=await page.locator('.mode-badge').innerText();if(!badge.includes(mode==='local'?'本地':'云端'))throw new Error('mode mismatch');await page.screenshot({path:root+'/screenshots/'+mode+'-home-desktop.png',fullPage:true});
await page.goto(base+'/courses',{waitUntil:'domcontentloaded'});if(await page.locator('.content-card').count()!==24)throw new Error('course pagination wrong');
if(mode==='local'){await page.goto(base+'/read/legacy-course-001',{waitUntil:'domcontentloaded'});await page.locator('.reader-body').waitFor();if((await page.locator('.reader-body').innerText()).length<100)throw new Error('missing local material');await page.screenshot({path:root+'/screenshots/local-reader.png'});const denied=await page.request.get(base+'/discover');if(denied.status()!==404)throw new Error('local discovery isolation failed');}else{await page.goto(base+'/discover',{waitUntil:'domcontentloaded'});await page.screenshot({path:root+'/screenshots/cloud-discover.png',fullPage:true});await page.goto(base+'/login',{waitUntil:'domcontentloaded'});if(!(await page.locator('main').innerText()).includes('GitHub'))throw new Error('cloud login missing');}
await page.setViewportSize({width:390,height:844});await page.goto(base,{waitUntil:'domcontentloaded'});if(await page.evaluate(()=>document.documentElement.scrollWidth>innerWidth+1))throw new Error('mobile overflow');await page.screenshot({path:root+'/screenshots/'+mode+'-home-mobile.png',fullPage:true});if(errors.length)throw new Error(errors.join(';'));results.push({mode,homeStatus:response.status(),modeBadge:badge,pagination:24,localReader:mode==='local',loginEntry:mode==='cloud',mobileOverflow:false,errors});await page.close();}await writeFile(root+'/browser.json',JSON.stringify(results,null,2)+'\n');console.log('Read-only browser verification passed for both modes; no account deletion or learning-state test writes.');}finally{await browser.close();}
