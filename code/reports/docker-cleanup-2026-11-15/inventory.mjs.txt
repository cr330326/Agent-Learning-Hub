import Database from '/workspace/code/node_modules/better-sqlite3/lib/index.js';
import {readFileSync,writeFileSync,readdirSync,statSync} from 'node:fs';
import {join} from 'node:path';
const names=JSON.parse(readFileSync('/work/volumes.json','utf8'));const result=[];
for(let i=0;i<names.length;i++){
 const root='/volumes/'+i,files=[];function walk(dir){for(const d of readdirSync(dir,{withFileTypes:true})){const p=join(dir,d.name);if(d.isDirectory())walk(p);else if(d.isFile())files.push({path:p.slice(root.length+1),bytes:statSync(p).size});}}walk(root);
 const databases=[];
 for(const f of files.filter(f=>/\.(?:sqlite|db)$/.test(f.path))){let db;try{db=new Database(join(root,f.path),{readonly:true,fileMustExist:true});db.pragma('busy_timeout=5000');const tables=db.prepare("SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' ORDER BY name").all().map(r=>r.name);const counts=Object.fromEntries(tables.map(t=>[t,db.prepare('SELECT COUNT(*) AS n FROM "'+t.replaceAll('"','""')+'"').get().n]));databases.push({path:f.path,bytes:f.bytes,counts,quickCheck:db.pragma('quick_check',{simple:true})});}catch(e){databases.push({path:f.path,error:e.code??e.message});}finally{db?.close();}}
 result.push({name:names[i],bytes:files.reduce((s,f)=>s+f.bytes,0),fileCount:files.length,databases,otherFileExtensions:[...new Set(files.filter(f=>!/\.(?:sqlite|db)(?:-wal|-shm|-journal)?$/.test(f.path)).map(f=>f.path.split('.').at(-1)))]});
}
writeFileSync('/reports/volume-inventory.json',JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify(result.map(v=>({name:v.name,mb:Math.round(v.bytes/1048576*10)/10,files:v.fileCount,databases:v.databases.map(d=>({path:d.path,users:d.counts?.users,notes:d.counts?.notes,items:d.counts?.items,quickCheck:d.quickCheck,error:d.error}))})),null,2));
